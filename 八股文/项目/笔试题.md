#include <iostream>
#include <set>
#include <unordered_map>
#include <algorithm>

using namespace std;

// 全局变量保存状态
int n, m;
set<int> seats; // 自动排序，从小到大存当前有人的座位号
unordered_map<int, int> people; // 记录 守护者ID -> 座位号

// 核心函数：找到当前最好的位置
int findBestSeat() {
    // 如果当前没人，直接坐第1个位置
    if (seats.empty()) {
        return 1;
    }

    int max_dist = -1;
    int best_pos = -1;

    // 1. 考虑坐在最左边 (位置 1)
    // 距离是：当前第一个人坐的位置 - 1
    int first_person = *seats.begin();
    int left_dist = first_person - 1;
    if (left_dist > max_dist) {
        max_dist = left_dist;
        best_pos = 1;
    }

    // 2. 考虑坐在两个人中间
    // 遍历 set，查看相邻两个人之间的空隙
    auto prev = seats.begin();
    auto curr = seats.begin();
    curr++; // curr 指向第二个人

    while (curr != seats.end()) {
        // 中间位置距离左右两边的距离是 (后-前)/2
        int dist = (*curr - *prev) / 2;
        int pos = *prev + dist; // 具体坐下的坐标

        // 如果距离比当前找到的更大，更新（如果距离一样，保留旧的，因为题目通常要求坐标更小的优先）
        if (dist > max_dist) {
            max_dist = dist;
            best_pos = pos;
        }

        prev = curr;
        curr++;
    }

    // 3. 考虑坐在最右边 (位置 n)
    // 距离是：n - 当前最后一个人坐的位置
    int last_person = *seats.rbegin();
    int right_dist = n - last_person;
    
    // 注意：这里必须是严格大于(>)。
    // 如果最右边的距离等于中间某个距离，我们要选坐标小的（中间那个），所以不更新。
    if (right_dist > max_dist) {
        max_dist = right_dist;
        best_pos = n;
    }

    return best_pos;
}

int main() {
    // 优化输入输出速度
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    cin >> n >> m;

    for (int i = 0; i < m; i++) {
        int t, d;
        cin >> t >> d;

        if (t == 1) {
            // t=1: 到达
            int pos = findBestSeat();
            
            // 输出位置
            cout << pos << "\n";
            
            // 更新状态
            people[d] = pos;
            seats.insert(pos);
            
        } else {
            // t=2: 离开
            int pos = people[d];
            
            // 输出位置
            cout << pos << "\n";
            
            // 更新状态
            people.erase(d);
            seats.erase(pos);
        }
    }

    return 0;
}

#include <iostream>
#include <vector>
#include <queue>
#include <cmath>

using namespace std;

struct Point {
    long long x, y;
};

// 状态结构体，放队列里用
struct Node {
    int id;    // 当前点的编号
    int step;  // 当前走了多少步
    int buff;  // 红药水剩余次数：-1代表没喝，0代表喝过且用完了，>0代表正在生效
};

int n, k;
long long r1, r2;
vector<Point> points;

// 计算两点距离的平方
long long distSq(int i, int j) {
    long long dx = points[i].x - points[j].x;
    long long dy = points[i].y - points[j].y;
    return dx * dx + dy * dy;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    if (!(cin >> n)) return 0;
    
    points.resize(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> points[i].x >> points[i].y;
    }
    
    // 输入 r1, k, r2
    // 注意：题目描述顺序可能有变，按你题目里的逻辑
    // "冲刺距离为r1... 剩下k次冲刺距离会变为r2"
    cin >> r1 >> k >> r2;

    long long r1_sq = r1 * r1;
    long long r2_sq = r2 * r2;

    // ==========================================
    // 阶段一：尝试不用红药水 (普通 BFS)
    // ==========================================
    vector<int> dist1(n + 1, -1);
    queue<int> q1;

    q1.push(1);
    dist1[1] = 0;

    while (!q1.empty()) {
        int u = q1.front();
        q1.pop();

        if (u == n) {
            // 成功到达！直接输出
            // 红药水0，蓝药水 = 步数
            cout << "0 " << dist1[n] << endl;
            return 0;
        }

        for (int v = 1; v <= n; v++) {
            if (dist1[v] == -1 && distSq(u, v) <= r1_sq) {
                dist1[v] = dist1[u] + 1;
                q1.push(v);
            }
        }
    }

    // ==========================================
    // 阶段二：必须用红药水 (带状态的 BFS)
    // ==========================================
    
    // visited[点编号][状态]
    // 状态定义映射：
    // index 0: 还没喝 (-1)
    // index 1 ~ k: 剩余次数 1~k
    // index k+1: 药效耗尽 (0)
    // 实际上我们可以用一个 map 或者稍微大一点的二维数组，为了简单，这里直接映射
    // buff 值域：-1 (没喝), 1~k (生效中), 0 (耗尽)
    
    // 我们用 vis[u][rem + 1] 来存储是否访问过，避免负数下标
    // rem = -1 -> 存下标 0
    // rem = 0  -> 存下标 1
    // rem = 1  -> 存下标 2 ... 以此类推
    vector<vector<bool>> visited(n + 1, vector<bool>(k + 2, false));
    queue<Node> q2;

    // 起点：在点1，步数0，buff -1 (没喝)
    q2.push({1, 0, -1});
    visited[1][0] = true; 

    while (!q2.empty()) {
        Node curr = q2.front();
        q2.pop();

        int u = curr.id;
        int step = curr.step;
        int buff = curr.buff;

        if (u == n) {
            // 找到了最短路
            // 红药水1，蓝药水 = 总步数
            // (通常题目里蓝药水指的是普通移动消耗，或者总移动消耗。如果红药水期间不耗蓝瓶，这里需要改一下逻辑)
            // 假设每一次移动都需要消耗一个瓶子(无论红蓝状态)
            cout << "1 " << step << endl;
            return 0;
        }

        for (int v = 1; v <= n; v++) {
            if (u == v) continue;
            long long d2 = distSq(u, v);

            // 分情况讨论状态转移

            // 1. 如果当前【没喝药】 (buff == -1)
            if (buff == -1) {
                // 选择 A: 继续不喝药，用 r1 走
                if (d2 <= r1_sq) {
                    if (!visited[v][0]) { // 0 对应 buff -1
                        visited[v][0] = true;
                        q2.push({v, step + 1, -1});
                    }
                }
                // 选择 B: 现在喝药！用 r2 走
                // 喝药后，本来剩 k 次，但这1次冲刺算用了1次，所以剩 k-1
                if (d2 <= r2_sq) {
                    int next_buff = k - 1; // 如果 k=1，这里变0
                    int idx = next_buff + 1;
                    if (!visited[v][idx]) {
                        visited[v][idx] = true;
                        q2.push({v, step + 1, next_buff});
                    }
                }
            }
            // 2. 如果当前【药效中】 (buff > 0)
            else if (buff > 0) {
                // 必须享受药效 (或者是只能走r2? 通常是可以走r2)
                if (d2 <= r2_sq) {
                    int next_buff = buff - 1;
                    int idx = next_buff + 1;
                    if (!visited[v][idx]) {
                        visited[v][idx] = true;
                        q2.push({v, step + 1, next_buff});
                    }
                }
            }
            // 3. 如果当前【药效耗尽】 (buff == 0)
            else { // buff == 0
                // 只能用 r1
                if (d2 <= r1_sq) {
                    if (!visited[v][1]) { // 1 对应 buff 0
                        visited[v][1] = true;
                        q2.push({v, step + 1, 0});
                    }
                }
            }
        }
    }

    // 如果阶段二也到不了
    cout << "-1 -1" << endl;

    return 0;
}
